Method 1: Python Script (Quick) ⏱️ 30 seconds
────────────────────────────────────────────────────────

from motor.motor_asyncio import AsyncIOMotorClient
import bcrypt

client = AsyncIOMotorClient("mongodb://localhost:27017")
db = client["bizpulse"]

user = {
    "email": "newuser@company.com",
    "password_hash": bcrypt.hashpw(b"Pass@123", bcrypt.gensalt()).decode(),
    "role": "sales",
    "access_level": {
        "businesses": ["Food"],           # ← CHOOSE FROM YOUR DATA
        "channels": ["Convenience"],      # ← CHOOSE FROM YOUR DATA
        "brands": ["Heinz"],              # ← CHOOSE FROM YOUR DATA
        "categories": ["*"],
        "data_types": ["revenue"]         # ← revenue / profit / costs
    },
    "perm_version": 1
}

db.users_permissions.insert_one(user)
print("✅ User created!")
